    <h2>Python WebSocket Chat Room</h2>
    <div id="chat-box"></div>
    
    <div id="input-container">
        <input type="text" id="message-input" placeholder="Type your message here..." autocomplete="off">
        <button onclick="sendMessage()">Send</button>
    </div>

    <script>
        // Prompt user for their name upon joining
        const username = prompt("Enter your username:", "User") || "Anonymous";
        
        // Connect to the Python WebSocket server
        const ws = new WebSocket("ws://localhost:6789");
        const chatBox = document.getElementById("chat-box");
        const messageInput = document.getElementById("message-input");

        // Handle incoming messages from the server
        ws.onmessage = function(event) {
            appendMessage(event.data);
        };

        // Handle errors or disconnections
        ws.onclose = function() {
            appendMessage("<i>System: Disconnected from the server.</i>");
        };

        function sendMessage() {
            const text = messageInput.value.trim();
            if (text !== "") {
                const fullMessage = `${username}: ${text}`;
                
                // Send text to the WebSocket server
                ws.send(fullMessage);
                
                // Immediately display your own message in your chat UI
                appendMessage(`<b>You:</b> ${text}`);
                messageInput.value = "";
            }
        }

        function appendMessage(msg) {
            const msgElement = document.createElement("div");
            msgElement.innerHTML = msg;
            chatBox.appendChild(msgElement);
            chatBox.scrollTop = chatBox.scrollHeight; // Auto-scroll to latest message
        }

        // Allow sending messages by pressing the "Enter" key
        messageInput.addEventListener("keypress", function(event) {
            if (event.key === "Enter") {
                sendMessage();
            }
        });
    </script>
